section Comparison

variable {R : Type u₁} {A : Type u₂} [Category.{v₁} R] [Category.{v₂} A]
variable [PCategoryStruct R] [PCategory R] [RestrictionCategory A]
variable [HasZeroMorphisms R] [RestrictionZero R]
variable [HasZeroMorphisms A] [RestrictionZero A]
variable [CoprodStruct R] [CategoryTheory.HasCoproducts R] [RestrictionCoproducts R]
variable [CoprodStruct A] [CategoryTheory.HasCoproducts A] [RestrictionCoproducts A]
variable [Ranges R] [Ranges A]

example (F : RangeFunctor R A) {U : R}
    (τ : nearProd U U ⟶ U) (hτ : RecursionCategory.IsTuringMorphism τ)
    (b : U ≅ coprod U U) (P N : Idem (F.toFunctor.obj U))
    (hPN : P ⊓ N = ⊥)
    (ha0 : F.mapIdem (Idem.pull (diag U ≫ τ ≫ b.hom) ((⊤ : Idem U).sum ⊥)) ≤ P)
    (ha1 : F.mapIdem (Idem.pull (diag U ≫ τ ≫ b.hom) ((⊥ : Idem U).sum ⊤)) ≤ N)
    (s : F.toFunctor.obj U ⟶ coprod (F.toFunctor.obj U) (F.toFunctor.obj U))
    (hs0 : Idem.pull s ((⊤ : Idem (F.toFunctor.obj U)).sum ⊥) = N)
    (hs1 : Idem.pull s ((⊥ : Idem (F.toFunctor.obj U)).sum ⊤) = P)
    (ℓ : U ⟶ coprod U U)
    (hℓ : F.toFunctor.map ℓ = s ≫ (F.comparison U U).hom) :

    ∃ g : U ⟶ U, isTotal g ∧ Idem.pull (F.toFunctor.map g) P = ⊥ ∧
      Idem.pull (F.toFunctor.map g) N = ⊥ ∧
      (F.toFunctor.map (𝟙 U) ≠ 0 →
        Idem.pull (F.toFunctor.map g) (P ⊔ N) =
          Idem.pull (F.toFunctor.map g) P ⊔ Idem.pull (F.toFunctor.map g) N → P ⊔ N ≠ ⊤) := by
  obtain ⟨g, hg, hp, hn⟩ := comparison_zero_inverse_images F τ hτ b P N hPN ha0 ha1 s hs0 hs1 ℓ hℓ
  exact ⟨g, hg, hp, hn, fun nz joins => comparison_noncoverage F g hg P N hp hn nz joins⟩

end Comparison

open FFL.Entailment NamedArithmeticSignature
example (A : SyntacticPredicates.Carrier) (pa : ArithmeticNamedAxiomComparison.ExtendsPA A)
    (cons : Entailment.Consistent (ArithmeticNamedProofConnectives.system A)) :
    ∃ φ : NamedArithmeticSignature.Formula,
      ¬ SyntacticNamedHilbert.Derives functionSymbols relationSymbols bound A (.neg φ) ∧
      ¬ SyntacticNamedHilbert.Derives functionSymbols relationSymbols bound A φ ∧
      φ.freeVariables = [] := by
  obtain ⟨φ,closed,pos,neg⟩ := ArithmeticSentenceInseparability.sentence_incomplete A pa cons
  exact ⟨φ,neg,pos,closed⟩
end

open OracleComputation
example (f : ℕ → ℕ ⊕ ℕ)
    (left : ∀ n, (∃ y : ℕ, Sum.inl y ∈ Part.some (f n)) ↔ n ∉ A)
    (right : ∀ n, (∃ y : ℕ, Sum.inr y ∈ Part.some (f n)) ↔ n ∈ A) : ¬ Computable f := by
  intro cf
  obtain ⟨_, impossible⟩ := Restricted.total_and_noncomputable_of_output_summands
    (fun n => Part.some (f n)) left right
  exact impossible ⟨f,cf,rfl⟩
end
